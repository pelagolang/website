//~ Shapes and their areas: enums, `switch`, structs and methods.

enum shape {
    circle(radius: float)
    rect(w: float, h: float)
    empty
}

fn area(s: shape) -> float {
    switch s {
        case .circle(r): 3.14159 * r * r
        case .rect(w, h): w * h
        case .empty: 0.0
    }
}

fn describe(s: shape) -> string {
    switch s {
        case .circle(r) if r > 10.0: "big circle"
        case .circle(_): "circle"
        case .rect(w, h): "rect #w by #h"
        case .empty: "nothing"
    }
}

struct point {
    x: float
    y: float

    fn length(self) -> float = (self.x * self.x + self.y * self.y).sqrt()

    fn shift(inout self, dx: float, dy: float) {
        self.x += dx
        self.y += dy
    }
}

fn main() {
    const shapes = [shape.circle(radius: 2.0), .rect(w: 3.0, h: 4.0), .empty]
    var total = 0.0
    for s of shapes {
        total += area(s)
        print(describe(s))
    }
    print("total: #total")

    var p = point { x: 3.0, y: 4.0 }
    p.shift!(1.0, 1.0)
    print(p.length())
}
